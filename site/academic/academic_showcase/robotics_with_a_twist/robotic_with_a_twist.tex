\noindent\underline{Homogeneous Transform \(\longrightarrow\) Twist}\\
Let \(\bm{T} =\begin{bmatrix}
    \bm{R} & \bm{d} \\ \bm{0}^T & 1
\end{bmatrix}\) known.\\
\textbf{if} \(\bm{R} = \bm{I}\) and \(\bm{d} \neq \bm{0}\):
\begin{equation*}
    \bm{\xi} = \begin{bmatrix}
        \bm{d} \\ \bm{0}
    \end{bmatrix} \hspace{30pt} \bm{\Xi} = \begin{bmatrix}
        \bm{0} & \bm{d} \\ \bm{0}^T & 0
    \end{bmatrix}
\end{equation*}
\textbf{elif} \(\bm{R} \neq \bm{I}\) and \(\bm{d} = \bm{0}\):

\hspace{10pt}Extract \(\bm{k}\) and \(\theta\) using angle-axis
\begin{equation*}
    \bm{\xi} = \begin{bmatrix}
        \bm{0} \\ \bm{k}
    \end{bmatrix}\theta \hspace{30pt} \bm{\Xi} = \begin{bmatrix}
        \bm{S}(\bm{k}\theta) & \bm{0} \\ \bm{0}^T & 0
    \end{bmatrix}
\end{equation*}
\textbf{else:}\\
\hspace{10pt}Extract \(\bm{k}\) and \(\theta\) using angle-axis
\begin{equation*}
    \bm{v} = \left[ \frac{1}{\theta} (\bm{I} - \bm{R})\bm{S}(\hat{\bm{k}}) + \hat{\bm{k}}\hat{\bm{k}}^T \right]^{-1}\bm{d}
\end{equation*}
\begin{equation*}
    \bm{\xi} = \begin{bmatrix}
        \bm{v} \\ \bm{k}
    \end{bmatrix}\theta \hspace{30pt} \bm{\Xi} = \begin{bmatrix}
        \bm{S}(\bm{k}\theta) & \bm{v} \\ \bm{0}^T & 0 
    \end{bmatrix}
\end{equation*}

\noindent\underline{Homogeneous Transform \(\longrightarrow\) Screw}\\
Let \(\bm{T} =\begin{bmatrix}
    \bm{R} & \bm{d} \\ \bm{0}^T & 1
\end{bmatrix}\) known.\\
Do HT \(\longrightarrow\) Twist, and get \(\bm{v}\), \(\bm{k}\), \(\theta\). Then 
\begin{equation*}
    h = (\bm{k}^T\bm{v})/\theta \hspace{30pt} \bm{q} =  (\bm{k}\times \bm{v})/ \theta
\end{equation*}

\noindent\underline{Screw \(\longrightarrow\) Twist}\\
Let \(\bm{q},\, \bm{k},\,h,\) and \(\theta\) known.\\
\textbf{if} \(h= \infty\):\\
\begin{equation*}
    \bm{v} = \bm{q} \times \bm{k} \hspace{10pt} \bm{\xi} = \begin{bmatrix}
        \bm{v} \\ \bm{0}
    \end{bmatrix} \hspace{10pt} \bm{\Xi} = \begin{bmatrix}
        \bm{0} & \bm{v} \\  \bm{0}^T & 0
    \end{bmatrix}
\end{equation*}
\textbf{else:}
\begin{equation*}
    \bm{\xi} = \begin{bmatrix}
        \bm{q} \times \bm{k} + \bm{k}h \\ \bm{k}
    \end{bmatrix} \theta = \begin{bmatrix}
        \bm{v} \\ \bm{k}
    \end{bmatrix}\theta  \hspace{20pt} \bm{\Xi} = \bm{\xi}^{\wedge}
\end{equation*}


\noindent\underline{Twist \(\longrightarrow\) Screw}\\
Let \(\bm{\Xi} = \begin{bmatrix}
    \bm{S}(\bm{k}\theta) & \bm{v} \\ \bm{0}^T & 0 
\end{bmatrix}\) known.\\
\hspace{10pt}Extract \(\bm{k}\), \(\theta\), and \(\bm{v}\) from \(\bm{\Xi}\)
\begin{equation*}
    h = (\bm{k}^T\bm{v})/\theta \hspace{30pt} \bm{q} =  (\bm{k}\times \bm{v})/ \theta
\end{equation*}

\noindent\underline{Screw \(\longrightarrow\) Homogeneous Transform}\\
Let \(\bm{q},\, \bm{k},\,h,\) and \(\theta\) known.\\
\textbf{if} \(h=\infty\):
\begin{equation*}
    \bm{T} = \begin{bmatrix}
        \bm{I} & \bm{v} \\ \bm{0}^T & 1
    \end{bmatrix}
\end{equation*}
\textbf{else:}
\begin{equation*}
    \bm{T} = \begin{bmatrix}
        \operatorname{exp}(\bm{S}(\bm{k}\theta)) & (\bm{I} - \operatorname{exp}(\bm{S}(\bm{k}\theta))\bm{q} + h\theta\bm{k} \\ \bm{0}^T & 1
    \end{bmatrix}
\end{equation*}

\noindent\underline{Twist \(\longrightarrow\) Homogeneous Transform}\\
Let \(\bm{\Xi} = \begin{bmatrix}
    \bm{S}(\bm{k}\theta) & \bm{v} \\ \bm{0}^T & 0 
\end{bmatrix}\) known.\\
\begin{equation*}
    \bm{T} = \operatorname{exp}(\bm{\Xi})
\end{equation*}
